"""Text resources from the widget wheel's same-origin public asset bundle."""
import json
import js


class Resource:
    def __init__(self, path):
        self.path = path

    @property
    def suffix(self):
        return '.' + self.path.rsplit('.', 1)[-1] if '.' in self.path else ''

    def iterdir(self):
        prefix = self.path + '/'
        return [Resource(path) for path in json.loads(js.pytinctureResourcePaths)
                if path.startswith(prefix) and '/' not in path[len(prefix):]]

    def read_text(self, encoding='utf-8'):
        if encoding.lower() != 'utf-8':
            raise ValueError('Browser resources use UTF-8')
        return js.pytinctureReadResource(self.path)


def files(package):
    return Resource(package.replace('.', '/'))
